window.processSelectedOperatorOffers = function() { var textInput = document.getElementById('ai-text-input'); var operatorInput = document.getElementById('selected-operator'); if (!textInput || !textInput.value.trim()) { alert('অনুগ্রহ করে অফারের টেক্সট পেস্ট করুন!'); return; } var text = textInput.value; var selectedOperator = operatorInput.value; var lines = text.split('\n'); var currentOffers = JSON.parse(localStorage.getItem('myOffers')) || []; var addedCount = 0; var badgeBg = "bg-danger text-white"; if (selectedOperator === "Banglalink") badgeBg = "bg-warning text-dark"; if (selectedOperator === "GP") badgeBg = "bg-info text-dark"; lines.forEach(function(line) { var trimmed = line.trim(); if (!trimmed) return; var bnDigits = ['০', '১', '২', '৩', '৪', '৫', '৬', '৭', '৮', '৯']; var convertedLine = trimmed.replace(/[০-৯]/g, function(d) { return bnDigits.indexOf(d); }); var priceMatch = convertedLine.match(/(\d+)\s*(টাকা|tk|bdt)/i) || convertedLine.match(/—\s*(\d+)/); if (!priceMatch) return; var price = priceMatch[1]; var daysMatch = convertedLine.match(/(\d+)\s*(দিন|days)/i) || ["", "৩০ দিন"]; var validity = daysMatch[0]; var cleanDetails = trimmed .replace(/^\d+[\.\)]\s*/, '') .replace(/—\s*[০-৯\d]+\s*টাকা.*/i, '') .replace(/—\s*[০-৯\d]+\s*(দিন|days).*/i, '') .trim(); currentOffers.push({ id: Date.now() + Math.random(), op: selectedOperator, title: cleanDetails, validity: validity, price: price, badgeBg: badgeBg }); addedCount++; }); localStorage.setItem('myOffers', JSON.stringify(currentOffers)); if (typeof renderAdminTable === 'function') { renderAdminTable(); } textInput.value = ''; alert(selectedOperator + " অপারেটরে মোট " + addedCount + " টি অফার যুক্ত হয়েছে!"); };
